# NODE 내장 모듈

## http 
서버 관련
~~~js
var http = require('http');
~~~

### http.createServer 
서버 생성?
~~~js
http.createServer([options][, requestListener])
~~~

#### 사용법
~~~js
var app = http.createServer(function(request, response){

})

// app.listen(포트번호)
app.listen(3000)
~~~

## fs
파일 관련 모듈
~~~js
var fs = require('fs');
~~~

### fs.writeFile
파일 생성
~~~js
fs.writeFile(file, data[, options], callback)
fs.writeFile(경로/파일이름, 내용[, utf8], function(error){
    if(error) throw error;
})

fs.writeFile(`폴더명/파일명`, '내용', 'utf8', function(err){

})
~~~

### fs.readdir 
폴더 내 파일 리스트를 보여준다
~~~js
fs.readdirSync(path[, options])
fs.readdirSync(폴더경로[, options])

fs.readdir('./data', function(error, filelist){

})
~~~

### fs.readFile
파일을 읽는다
~~~js
fs.readFile(path[, options], callback)
fs.readFile(파일경로[, utf8], callback)

fs.readFile(`data/${filteredId}`, 'utf8', function(err, description){
    // description 내용이 담겨있는 변수
})
~~~

### fs.rename
파일 이름을 변경한다
~~~js
fs.rename(oldPath, newPath, callback)
fs.rename(경로/파일이름, 경로/새로운파일이름, callback)

fs.rename(`data/${id}`, `data/${title}`, function(error){
    
})
~~~

## querystring
~~~js
var qs = require('querystring');
~~~

### qs.parse
URL 쿼리 문자열을 키와 값(obj)으로 변환해준다
~~~js
querystring.parse(str[, sep[, eq[, options]]])

querystring.parse(데이터);
~~~
