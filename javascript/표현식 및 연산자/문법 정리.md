# 문법 정리

## 구조 분해 할당
__구조 분해 할당__ 구문은 배열이나 객체의 속성을 해체하여 그 값을 개별 변수에 담을 수 있게 하는 JavaScript 표현식입니다.

~~~js
let a, b, rest;
[a, b] = [10, 20];

console.log(a);
// 예상 출력: 10

console.log(b);
// 예상 출력: 20

[a, b, ...rest] = [10, 20, 30, 40, 50];

console.log(rest);
// 예상 출력: Array [30, 40, 50]
~~~

## spread ( 확산 ) 연산자
spread 연산자 를 사용하면 기존 배열이나 객체의 전체 또는 일부를 다른 배열이나 객체로 빠르게 복사할 수 있습니다.
~~~js
const array = [1,2,3,4]
const thinCopy = array
const deepCopy = [...array]
thinCopy[0] = 5

console.log(thinCopy)
// 예상 출력: Array [5,2,3,4]
console.log(array)
// 예상 출력: Array [5,2,3,4]
console.log(thinCopy === array);
// 예상 출력: true

deepCopy[0] = 6
console.log(deepCopy)
// 예상 출력: Array [6,2,3,4]
console.log(array)
// 예상 출력: Array [5,2,3,4]
// thinCopy에서 변경 했기 때문에 처음 값이 5
console.log(deepCopy === array);
// 예상 출력: false

// 스프레드 연산자를 사용하면 깊은 복사를 해서 독립적인 배열을 만들 수 있다
~~~
