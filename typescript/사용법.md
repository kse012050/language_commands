# 타입스크립트 사용법

~~~js
/* 
    - type
    string
    number
    boolean
    null
    undefined
    bigint
    []
    {} 
    등
*/

let 문자 : string = 'kim';
// 문자 : string 

let 배열1 : string[] = ['kim' , 'park']
// 배열 : array 내용은 string 

let 오브젝트1 : {name : string} = { name : 'kim'}
// 오브젝트1 : object 안 name 속성은 string

let 오브젝트2 : {name? : string} = { name : 'kim'}
// 오브젝트1 : object 안 name 속성은 string 이지만 name이 없을 수도 있다고 표시(?)

let 문자혹은숫자1 : string | number = 'kim'
문자혹은숫자1 = 123;
// 문자혹은숫자 : "문자" 혹은 숫자

type 문자혹은숫자_타입 = string | number;
// type만 저장하는 변수?를 따로 만들 수 있다
// * 보통 카멜케이스 (첫글자 대문자) 로 많이 만든다

let 문자혹은숫자2 : 문자혹은숫자_타입 = 'kim';
문자혹은숫자2 = 123;


function 함수1(x : number)  {
    // 함수(매개변수 타입) : 리턴값 
    // 리턴 값이 없을 땐 , 비워놓거나 : void ( return을 막아 줌 )
    x * 2
}

function 함수2(x : number) : number {
    // 함수(매개변수 타입) : 리턴값 타입
    return x * 2
}

// function 함수3(x? : number) : number {
function 함수3(x : number | undefined) : number {
    // x? 는 들어올 수도 있고 안들어 올 수도 있다는 표현
    // x? 또는 | undefined 같은 의미
    return x * 2
}


function 함수4(x : number | string) {
    if(typeof x == 'number'){
        return x * 2
        // 8
    }else{
        return 'x' + x;
        // xx
    }
}

함수4(4)


type 배열_타입1 = [number , boolean];
// 배열의 첫번째 값은 무조건 number 두번째는 boolean

let 배열2 : 배열_타입1 = [123,true]
// 배열_타입 이 [숫자 , 불리언] 이기 때문에 형식을 맞춰야 한다

type 배열_타입2 = (number | string)[];
// 배열 안에는 숫자 또는 문자
// () 가 없으면  숫자 또는 배열 일 경우 문자만

let 배열3 : 배열_타입2 = [123,'kim']
// 배열_타입 이 [숫자 , 불리언] 이기 때문에 형식을 맞춰야 한다

type 오브젝트_타입 = {
    [key : string] : string
    // 오브젝트의 모든 속성은 string
}
let 오브젝트3 : 오브젝트_타입 = {name : 'kim'}


let 아무거나 : any;
// 아무거나 넣는 건데... 이걸 쓰면 typescript 쓰는 의미가 없어짐
아무거나 = 123;
아무거나 = 'kim';
아무거나 = [];

let any보다안전 : unknown;
// any 타입과 똑같이 아무거나 넣을 수 있다
any보다안전 = 123;
any보다안전 = 'kim';
any보다안전 = [];

let anyUnknown : string = 아무거나;
// any 와 unknown의 차이는 다른 변수에 넣을 때
// 타입에 따라 에러를 발생시켜주냐 안주냐의 차이다
// any 에러 X , unknown 에러 O
~~~
